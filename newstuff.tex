\section{This is the new stuff}


The following lemma is well-known (see, for example \cite{}).

\begin{lem}\label{bag_separator}
  Let $T$ be a tree and let $\omega:V(T)\to\R^+$ be a weight function.  Then, for any $b\in\N^+$, there exists $S\subseteq V(T)$ with $|S|\le b-1$ such that $\omega(C)\le \omega(T)/b$ for each component $C$ of $T-S$.
\end{lem}

\begin{proof}
  The proof is by induction on $b$.  If $b=1$, then $S:=\emptyset$ satisfies the requirements of the lemma.  Now assume $b\ge 2$. Root $T$ at an arbitrary vertex. Let $x\in V(T)$ be such that $\omega(T_x) \ge \omega(T)/b$ but $\omega(T_y) < \omega(T)/b$ for each child $y$ of $x$.  Then $T-x$ has at most one component $T_0$ with $\omega(T_0) > \omega(T)/b$.  If no such component exists, then $S:=\{x\}$ satisfies the requirements of the lemma and we are done.  Otherwise, the component $T_0$ has $\omega(T_0)\le (1-1/b)\omega(T)$.  Applying induction on $T_0$ with the weight function $\omega|_{V(T_0)}$ and $b-1$ yields a set $S_0$ of size at most $b-2$ such that $\omega(C)\le\omega(T_0)/(b-1)\le \omega(T)/b$ for each component $C$ of $T_0-S_0$.  The set $S:=S_0\cup\{x\}$ satisfies the requirements of the lemma.
\end{proof}

\begin{lem}
  Let $G$ be an $n$-vertex graph and let $\mathcal{T}=(T,(B_x\mid x\in V(T)))$ be a tree-decomposition of $G$.  Then, for any integer $b\in\N^+$,  $G$ has a tree-decomposition $\mathcal{Q}:=(Q,(C_q\mid q\in V(Q)))$ with the following properties:
  \begin{enumerate}[nosep,nolistsep,label=\rm(\alph*)]
    \item\label{req:height} $\height(Q)\le \log_b n$;
    \item\label{req:subtree_decomposition} for each $q\in V(Q)$, the torso $\torso{G}{\mathcal{Q},C_q}$ has a tree-decomposition $\mathcal{T}(q):=(T(q),(D_x:x\in V(T(q))))$ such that
    \begin{enumerate}[nosep,nolistsep,label=\rm(\roman*)]
      \item\label{req:subtree_size} $|V(T(q))|\le 4b-1$;
      \item\label{req:subtree_adhesion} each adhesion of $\mathcal{T}(q)$ is contained in the union of at most three adhesions of $\mathcal{T}$;
      \item\label{req:subtree_bags} each bag of $\mathcal{T}(q)$ is contained in the union of three adhesions of $\mathcal{T}$ or the union of one bag of $\mathcal{T}$ and one adhesion of $\mathcal{T}$.
    \end{enumerate}
  \end{enumerate}
\end{lem}

\begin{proof}
  The proof is by induction, with a slightly more complicated inductive hypothesis. An inductive instance $(G,\mathcal{T}, \alpha, \beta)$ involves a graph $G$, a tree-decomposition $\mathcal{T}:=(T,(B_x\mid x\in V(T)))$, and two (not necessarily distinct) leaves $\alpha$ and $\beta$ of $T$.  We will produce a tree-decomposition $\mathcal{Q}=(Q,(C_q\mid q\in V(Q)))$ of $G$ that satisfies \ref{req:subtree_decomposition} as well as the following two conditions, the first of which is a strengthening of \ref{req:height}:
  \begin{enumerate}[nosep,nolistsep,label=\rm(\alph*')]
    \item\label{req:height_prime} $\height(Q)\le \log_b (n-|B_\alpha \cup B_\beta|)$;
  \end{enumerate}
  \begin{enumerate}[nosep,nolistsep,label=\rm(\roman*)]\setcounter{enumi}{3}
    \item\label{req:adhesion_pairs} for the root $q_0$ of $Q$, there exists a bag of $\mathcal{T}(q_0)$ that contains $A_\alpha\cup A_\beta$, where $A_\alpha$ and $A_\beta$ are the adhesions of $\mathcal{T}$ incident to $\alpha$ and $\beta$ respectively.
  \end{enumerate}
  The induction is on the number of vertices in $V(G)\setminus (B_\alpha\cup B_\beta)$.  To see that this is enough to establish the original lemma, we can attach two leaves $\alpha, \beta$ to $T$, set $B_\alpha:=B_\beta:=\emptyset$, and apply the refined result.
  
  Define the weight function $\omega: V(T)\to\R^+$ by $\omega(x):=|B_x\setminus (A_x\cup B_\alpha\cup B_\beta)|$, where $A_x$ is the parent adhesion of $x$ in $\mathcal{T}$.  Apply \cref{bag_separator} to $T-\{\alpha,\beta\}$ with weight function $\omega$ to obtain a set $S_0$ of at most $b-1$ nodes of $T$ such that $\omega(C)\le\omega(T)/b=(n-|B_\alpha\cup B_\beta|)/b$, for each component $C$ of $T-S_0$.  Let $T_0$ be the minimum subtree of $T$ that spans $S_0\cup\{\alpha,\beta\}$. Let $S_1$ be the set of nodes of $T_0$ that have degree at least $3$ in $T_0$.  Then $|S_1|\le |S_0|\le b-1$. Let $S:=S_0\cup S_1\cup\{\alpha,\beta\}$, so $|S|\le 2b$.

  The tree $Q$ in our tree-decomposition $\mathcal{Q}=(Q,(C_x:x\in V(Q)))$ will have root $q_0$ with $C_{q_0}:=\bigcup_{x\in S}B_x$.  We now describe the tree-decomposition $\mathcal{T}(q_0):=(T(q_0), D_x:x\in V(T(q_0)))$.  First we describe the tree $T(q_0)$.  Say that a pair of nodes $x,y\in S$ is \defin{consecutive} if the path in $T$ from $x$ to $y$ has no internal node in $S$. For each pair of consecutive nodes $x,y\in S$, introduce a vertex $v(xy)$ and add the $3$-node path $x,v(xy),y$ to $T(q_0)$.  For each $x\in S$, set $D_x:=B_x$.  This gives a tree $T(q_0)$ with $2|S|-1\le 4b-1$ nodes.  For each pair of consecutive nodes $x,y\in S$ with $xy\in E(T)$, we set $D_{v(xy)}:=B_x\cap B_y$.  If $xy\not\in E(T)$, then $T-S$ has a component $C$ with $N_T(V(C))=\{x,y\}$, and $D_{v(xy)}$ will be defined in the next paragraph. 

  We now construct the tree-decomposition $\mathcal{T}(q_0)$ by examining the interaction of each component $C$ of $T-S$ with $S$.  Let $C$ be a component of $T-S$ and let $N:=N_T(V(C))$.  Since $S$ is non-empty, $|N|\ge 1$.  By the choice of $S_1$, $|N|\le 2$.  Let $\{\alpha',\beta'\}=N_T(V(C))$, so $\alpha'$ and $\beta'$ are consecutive vertices of $S$. (Possibly $\alpha'=\beta'$ if $|N|=1$.)  Then $T$ contains edges $x'\alpha'$ with $x'\in V(C)$ and $y'\beta'$ with $y'\in V(C)$.  Let $A_{\alpha'}:=B_{x'}\cap B_{\alpha'}$ and let $A_{\beta'}:=B_{y'}\cap B_{\beta'}$.  If $\alpha'=\beta'$, there is no need to do anything else.  If $\alpha'\neq\beta'$, then $\alpha'$ and $\beta'$ are two consecutive nodes in $S$.  In this case, we set $D_{v(\alpha'\beta')}:=A_{\alpha'}\cup A_{\beta'}$. 

  It is straightforward to check that $\mathcal{T}(q_0)$ is a tree-decomposition of $G[C_{q_0}]$.  We now argue that $\mathcal{T}(q_0)$ is a tree-decomposition of $\torso{G}{\mathcal{Q},C_{q_0}}$ as well.  The only danger is that $\torso{G}{\mathcal{Q},C_{q_0}}$ contains an edge $vw$ that is not an edge of $G[C_{q_0}]$.  Any such edge comes from a component $C$ of $T-S$ that creates a child $q_C$ of $q_0$ in $Q$ with $N_T(V(C)) = \{\alpha', \beta'\}$ and $\{v,w\}\subseteq A_{\alpha'}\cup A_{\beta'}$.  Indeed, $A_{\alpha'}\cup A_{\beta'}$ is a clique in $\torso{G}{\mathcal{Q},C_{q_0}}$, so it must appear in some bag of $\mathcal{T}(q_0)$ and in some bag of $\mathcal{T}(q_C)$.  If $\alpha'\neq\beta'$, then while processing $C$ we set $D_{v(\alpha'\beta')}:=A_{\alpha'}\cup A_{\beta'}$ above, which ensures that some bag of $\mathcal{T}(q_0)$ contains both $v$ and $w$. If $\alpha'=\beta'$, then $\{v,w\}\subseteq A_{\alpha'}\subseteq B_{\alpha'}=D_{\alpha'}$, which is a bag of $\mathcal{T}(q_0)$.  Thus $\mathcal{T}(q_0)$ is a tree-decomposition of $\torso{G}{\mathcal{Q},C_{q_0}}$. After processing each component $C$ of $T-S$ there is a final step:  
  
  To ensure that $\mathcal{T}(q_0)$ satisfies \ref{req:adhesion_pairs} we add $A_{\alpha}$ to $D_x$ for each $x\in V(T(q_0))$ on the path from $\alpha$ to $\beta$.  This does not change the fact that $\mathcal{T}(q_0)$ is a tree-decomposition of $\torso{G}{\mathcal{Q},C_{q_0}}$.  This completes the definition of $\mathcal{T}(q_0)$.
  We now verify that $\mathcal{T}(q_0)$ satisfies \ref{req:subtree_size}, \ref{req:subtree_adhesion}, \ref{req:subtree_bags}, and \ref{req:adhesion_pairs}.  Since $|V(T(q_0))|\le 4b-1$, \ref{req:subtree_size} is satisfied. Each edge of $T(q_0)$ has an endpoint $x\in V(T)$ and an endpoint $v(xy)\not\in V(T)$.  Thus, each adhesion of $\mathcal{T}(q_0)$ is contained in $D_{v(xy)}$ for some consecutive pair $x,y\in S$. Such a bag $D_{v(xy)}$ may contain $A_{\alpha}$, which is an adhesion of $\mathcal{T}$ (if $v(xy)$ is on the path from $\alpha$ to $\beta$ in $T(q_0)$).  In addition to this, $D_{v(xy)}$ contains either a single adhesion $B_x\cap B_y$ of $\mathcal{T}$ (if $xy\in E(T)$), or two adhesions $A_{\alpha'}\cup A_{\beta'}$ of $\mathcal{T}$ (when $xy\not\in E(T)$). Thus, each adhesion of $\mathcal{T}(q_0)$ is the union of at most three adhesions in $\mathcal{T}$, so $\mathcal{T}(q_0)$ satisfies \ref{req:subtree_adhesion}.  For each node $v(xy)\in V(T(q_0))\setminus V(T)$, the bag $D_{v(xy)}$ is the union of at most three adhesions of $\mathcal{T}$, as we have just discussed.  For each node $x\in V(T(q_0))\cap V(T)$, $D_x\subseteq B_x\cup A_\alpha$. Therefore $\mathcal{T}(q_0)$ satisfies \ref{req:subtree_bags}.  Finally, $D_\beta=A_\alpha\cup B_\beta\supseteq A_\alpha\cup A_\beta$, so $\mathcal{T}(q_0)$ satisfies \ref{req:adhesion_pairs}.
  
  We have defined the root bag $C_{q_0}$ of $\mathcal{Q}$ and the tree-decomposition $\mathcal{T}(q_0)$ of $\torso{G}{\mathcal{Q},C_{q_0}}$.  We now describe the rest of the tree-decomposition $\mathcal{Q}$ by setting up an inductive instance for each component $C$ of $T-S$ with $\omega(C) \ge 1$.  Let $C$ be such a component with $N_T(V(C)) = \{\alpha',\beta'\}$ (possibly $\alpha'=\beta'$).
  Let $T':=T[V(C)\cup\{\alpha',\beta'\}]$ and let $G_{\alpha'\beta'}:=G[\bigcup_{x\in V(T')}B_x]$.  Then $\mathcal{T}':=(T',(B_x:x\in V(T')))$ is a tree-decomposition of $G_{\alpha'\beta'}$.  Furthermore $G_{\alpha'\beta'}-(B_{\alpha'}\cup B_{\beta'})$ has $\omega(C)\le (n-|B_{\alpha}\cup B_{\beta}|)/b$ vertices.  Thus $(G_{\alpha'\beta'},\mathcal{T}',\alpha',\beta')$ is an instance on which we can apply induction.  The result of this induction is a tree-decomposition $\mathcal{Q}_{\alpha'\beta'}:=(Q_{\alpha'\beta'},(C_x\mid x\in V(T_{\alpha'\beta'})))$ of height at most $\log_b((n-|B_{\alpha}\cup B_{\beta}|)/b)=\log_b(n-|B_{\alpha}\cup B_{\beta}|)-1$.  Let $q'_0$ be the root of $Q_{\alpha'\beta'}$.  The tree $Q$ contains the edge $q_0q'_0$.
  
  By induction, there exists a tree-decomposition $\mathcal{T}(q'_0)$ of $\torso{G}{\mathcal{Q}_{\alpha'\beta'},C_{q'_0}}$ that satisfies \ref{req:subtree_size}, \ref{req:subtree_adhesion}, \ref{req:subtree_bags}, and \ref{req:adhesion_pairs}.  We claim that $\mathcal{T}(q'_0)$ is also a tree-decomposition of $\torso{G}{\mathcal{Q},C_{q'_0}}$. Indeed, the only difference between $\torso{G}{\mathcal{Q}_{\alpha'\beta'},C_{q'_0}}$ and $\torso{G}{\mathcal{Q},C_{q'_0}}$ is that the latter has additional edges between the vertices of $A_{\alpha'}\cup A_{\beta'}$. However, \ref{req:adhesion_pairs} ensures that some bag of $\mathcal{T}(q'_0)$ contains $A_{\alpha'}\cup A_{\beta'}$.
\end{proof}

\begin{lem}
  Let $\mathcal{G}$ be a hereditary class of graphs that admits a weighted $(g_1,g_2,g_3)$ mixed labelling scheme where $g_i:\N\to\R$ is a non-decreasing function with $g_i(n)\in\Oh(\log n)$ for each $i\in[3]$.   Let $J(v):=\{i\in\{0,\ldots,r-1\} : N_G(v)\cap K_i\neq\emptyset\}$.
  The bitstring $\tau(v)$ will be used to store $\mu_{x_i}(K_i)$ for each $i\in J(v)$.  However, there are some $i\in J(v)$ such that $\mu_{x_i}(K_i)$ is contained in $\id(x_r)=\id(x)$. This occurs precisely when $x_{i}x_{i+1}\in Y$. Thus, we define

    \tau(v) := \left\langle \langle \pi(x_i), \mu_{x_i}(K_i) \rangle \middle| i\in J(v),\; x_{i}x_{i+1}\notin Y \right\rangle \enspace .
  \]
  Thus, $\tau(v)$ contains at most $k$ clique labels.  Consider some $i\in J(v)$ such that $x_{i}x_{i+1}\notin Y$ and let $x$ be the maximum depth ancestor of $x_{i+1}$ that is in $X$. Thus, $x$ and $x_i$ are in the same component of $T-Y$. By \ref{Y_not_too_fast}
  \begin{align*}
    |\pi(x_i)|+|\mu_{x_i}(K_i)| 
    & \le \delta(x)-\log\min_{u\in K_i}\delta(u) + \Oh(g_3(n)+\log\log n) \\

    & \le \log b + \Oh(g_3(n)+\log\log n) 
  \end{align*}
  Thus, $|\tau(v)|\le \Oh(k(\sqrt{\log b} + g_3(n) + \log k))$.  
  
  We now describe the parts of $\alpha(v)$.
  For each $u\in N_G(v)\cap A_x$, exactly one of the following cases applies:
  \begin{enumerate}[nosep,nolistsep]

    \item $\mu_{h(u)}(K_u)$ is contained in $\tau(v)$.  In this case, we define $b(v,u)=1$ and define $i(v,u)$ to be the index of $\mu_{h(u)}(K_u)$ in the sequence $\tau(v)$. \pat{Think about this......}  We also define $q(v,u):=\pi_{F_j}(x_j)$ $x_j=h(u)$.

  \paragraph{\boldmath The vertex label length $|\mu(v)|$:}
  We now consider the length of the vertex label $\mu(v)$ for a vertex $v\in B_x\setminus A_x$. This label contains $\id(x)$ and $\id(h(u))$ for each $u\in N_G(v)\cap A_x$.  These identifiers have total length at most $(k+1)\lceil\log b\rceil$, so these contribute at most $\Oh(k\log b)$ bits to the length of $\mu(v)$.  In addition to this, $\alpha(v)$ contains up to $k$ pairs, each of which has length $\Oh(\log k + g_2(n))$.  The only remaining parts of $\mu(v)$ are $\mu_x(v)$ and the clique labels in $\tau(v)$. Let $J:=\{i\in\{0,\ldots,r-1\} \mid N_G(v)\cap B_{x_i}\setminus A_{x_i}\neq\emptyset\}$. The clique labels in $\tau(v)$ are exactly $\mu_{x_i}(K_i)$ for each $i\in J$. Since $|J|\le k$, \eqref{telescoping_subset} implies
  \begin{align*}
    \sum_{i\in J} |\mu_{x_i}(K_i)| 
    & \le \sum_{i\in J}\left( \log(\delta(B_{x_i}\setminus A_{x_i})) - \log \min_{w\in K_i}\delta(w) + g_3(n) \right) \\    
    & \le \log\omega(G)-\log\omega(x)+|J|\log k + |J|g_3(n) \\
    & \le \log\omega(G)-\log\omega(x)+\Oh(k\cdot (\log k+ g_3(n))) \enspace .
  \end{align*}
  Therefore,
  \begin{align*}
    \sum_{i\in J} |\mu_{x_i}(K_i)| + |\mu_x(v)| 
    &\le \sum_{i\in J} |\mu_{x_i}(K_i)| + \log\delta(B_x\setminus A_x)-\log\delta(v) + g_1(n) \\ 
    &\le \sum_{i\in J} |\mu_{x_i}(K_i)| + \log\omega(x)-\log\omega(v) + g_1(n) + \Oh(\log k) \\ 
     & \le \log\omega(G)-\log\omega(v)+g_1(n)+\Oh(k\cdot (g_3(n)+\log\log n)).
  \end{align*}
  Thus $\mu(v)$ has total length at most $\log\omega(G)-\log\omega(v)+g_1(n)+\Oh(k\cdot(\log b+g_2(n)+g_3(n)+\log\log n))$, which satisfies the requirement for $g'_1(n)$.

  \paragraph{Adjacency testing:}  We now describe the adjacency tester $A'$ that takes two labels $\mu(v)$ and $\mu(w)$.  Suppose that $v\in B_x\setminus A_x$ and $w\in B_y\setminus A_y$ for some pair of node $x,y\in V(T)$.  Thus $\mu(v)=\langle \id(x),\mu_x(v),\tau(v),\alpha(v)\rangle$ and $\mu(w)=\langle \id(y),\mu_y(w),\tau(w),\alpha(w)\rangle$.  The adjacency tester first checks if $\id(x)=\id(y)$.  If $\id(x)=\id(y)$ then $x=y$, since $\id$ is injective. In this case $vw\in E(G)$ if and only if $vw\in E(G[B_x\setminus A_x])$, so $A'(\mu(v),\mu(w)):=A(\mu_x(v),\mu_y(w))$.

  If $\id(x)\neq \id(y)$ then the only way in which $v$ and $w$ can be adjacent in $G$ is if $w\in N_G(v)\cap A_x$ or $v\in N_G(w)\cap A_y$.  We now explain how to test if $v\in N_G(w)\cap A_y$. For each $u\in N_G(w)\cap A_y$, $\alpha(w)$ contains an entry $\langle i(w,u),\kappa_{h(u)}(K_u,u)\rangle$ where $K_u$ is a clique in $\torso{G^+}{\mathcal{T},B_{h(u)}}-A_{h(u)}$ that contains $u$ and $i(w,u)$ is the index of $\langle\id(h(u)),\mu_{h(u)}(K_u)\rangle$ in $\tau(w)$.  Thus, the adjacency tester $A'$ examines each entry $\langle i(w,u),\kappa_{h(u)}(K_u,u)\rangle$.  For each entry, it checks if $\id(h(u))=\id(x)$.  If not, then $h(u)\neq x$, $u\in B_{h(u)}\setminus A_{h(u)}$ and $v\in B_x\setminus A_x$, so $u\neq v$.  Otherwise, $x=h(u)$ and the tester checks if $I(\mu_{h(u)}(K_u),\kappa_{h(u)}(K_u,u),\mu_x(v))=1$.  If so, then $u=v$ so $v\in N_G(w)\cap A_y$.  In particular, $v\in N_G(w)$ so $A'(\mu(v),\mu(w)):=1$.  By doing this for each entry $\langle i(w,u),\kappa_{h(u)}(K_u,u)\rangle$ in $\alpha(w)$ the adjacency tester either returns $1$ because $v\in N_G(w)\cap A_y$ or determines that $v\not\in N_G(w)\cap A_y$.  The tester then repeats this process with the roles of $v$ and $w$ reversed and either returns $1$ because $w\in N_G(v)\cap A_x$ or determines that $w\not\in N_G(v)\cap A_x$.  In the latter case, $A'(\mu(v),\mu(w)):=0$ since $vw\not\in E(G[B_x\setminus A_x])$, $v\not\in N_G(w)\cap A_y$ and $w\not\in N_G(v)\cap A_x$.

  \paragraph{The clique label $\mu(K)$:}
  Let $K$ be a clique in $G^+$.  Consider the set $h(K):=\{x\in V(T):K\cap B_x\setminus A_x\neq\emptyset\}$.  Since $K$ is a clique in $G^+$, the nodes in $h(K)$ lie on a single path in $x_0,\ldots,x_r$ from the root $x_0$ of $T$ to the node $x_r\in h(K)$ of maximum $T$-depth. For each  $x\in h(K)\setminus \{x_r\}$, the adhesion $A_{x_r}$ contains at least one vertex in $B_x\setminus A_x$. Therefore, $|h(K)|\le|A_{x_r}|+1\le k+1$.  For each $x\in h(K)$, $K\cap B_x\setminus A_x$ is a non-empty clique in $\torso{G^+}{\mathcal{T},B_x}-A_x$, so $\mu_x(K\cap B_x\setminus A_x)$ is defined. 
  Order the vertices of $A_{x_r}$ in some arbitrary but fixed order $a_1,\ldots,a_{|A_{x_r}|}$.  Let $c(K)$ denote the length-$|A_{x_r}|$ bit-vector whose $i$th bit indicates if $a_i\in K$.  Define the clique label
  \[
    \mu(K)
    :=\left\langle \left\langle \mu_x(K\cap B_x\setminus A_x)
    \mid x\in h(K)
    \right\rangle, c(K) \right\rangle \enspace .
  \]

  We now upper bound the length of $\mu(K)$. Since $\mathcal{T}$ has adhesion-width at most $k$, $|c(K)|\le k$.  Just like the clique labels that appear in vertex labels, the lengths of the at most $k+1$ clique labels in the first part of $\mu(K)$ form a telescoping series so that
  \begin{align*}
    \sum_{x\in h(K)}|\mu_{x}(K\cap B_x\setminus A_x)|
     & \le
    \log\omega(G)-\log\min_{u\in K\cap B_z\setminus A_z}(\omega(u)) + \Oh(k\cdot(g_3(n)+\log k)) \\
     & \le \log\omega(G)-\log\min_{u\in K}(\omega(u))
    +\Oh(k\cdot(g_3(n)+\log k))
  \end{align*}
  Thus, $|\mu(K)|\le \log\omega(G)-\log\min_{u\in K}(\omega(u))
    +\Oh(k\cdot(g_3(n)+\log\log n))
  $, which satisfies the requirement for $g'_3(n)$.

  The length of $\kappa(K,u)$ is $g_2(n)+\Oh(\log b+\log k+\log\log n)$, which satisfies the requirement for $g'_2(n)$.
